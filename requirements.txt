# Core runtime: wake word (Vosk), transcription (Whisper), speech (Piper).
#
# Installed into a uv-managed interpreter under ~/.local/share/agentvoice/venv,
# deliberately NOT the system Python. Arch is a rolling release -- uv already
# offers CPython 3.15 -- and a venv built on /usr/bin/python breaks the next
# time Arch bumps the interpreter, which would leave voice silently dead after
# an `omarchy update`.
#
# vosk on PyPI is a py3 wheel rather than a CPython-version-specific one, so
# nothing here ties us to a particular interpreter release.

vosk>=0.3.45,<0.4
faster-whisper>=1.2,<2
piper-tts>=1.8,<2
sounddevice>=0.5,<1
numpy>=2,<3
